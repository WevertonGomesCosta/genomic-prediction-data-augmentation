1. Objetivo

Os modelos GBLUP utilizam a semelhança genômica entre os indivíduos. Essa informação é resumida em uma matriz chamada G.

Cada elemento de G representa o grau de relacionamento genômico entre dois indivíduos. Indivíduos com padrões de SNPs mais semelhantes tendem a apresentar maior relacionamento genômico.

Neste projeto, G é calculada uma única vez usando todos os indivíduos genotipados. Os fenótipos de validação continuarão mascarados durante cada ajuste; portanto, usar os genótipos dos indivíduos de validação em G não fornece seus fenótipos ao modelo.

3. Frequência alélica

Como os SNPs estão codificados como 0, 1 e 2, a média de uma coluna corresponde a duas vezes a frequência do alelo contado. Assim, para cada marcador:

\[ p_j = \frac{\bar M_j}{2}. \]

p <- colMeans(M) / 2

summary(p)
   Min. 1st Qu.  Median    Mean 3rd Qu.    Max. 
 0.4275  0.5685  0.6186  0.6210  0.6730  0.9007

4. Centralização dos marcadores

A contribuição esperada do marcador j é 2p_j. Subtraímos esse valor de cada genótipo para obter a matriz centralizada Z:

\[ Z_{ij}=M_{ij}-2p_j. \]

Z <- sweep(
  M,
  2,
  2 * p,
  FUN = "-"
)

dim(Z)
[1] 1379 4325

5. Construção da matriz G

Utilizamos a forma aditiva de VanRaden:

\[ \mathbf{G}= \frac{\mathbf{Z}\mathbf{Z}^{\mathsf{T}}} {2\sum_j p_j(1-p_j)}. \]

denominador <- 2 * sum(
  p * (1 - p)
)

G <- tcrossprod(Z) / denominador

Adicionamos um valor numérico muito pequeno à diagonal para evitar problemas de estabilidade computacional sem alterar de forma relevante os relacionamentos.

diag(G) <- diag(G) + 1e-6

9. Relacionamento máximo e cobertura genômica

O script original também verificava quão próximo cada indivíduo da validação estava dos indivíduos disponíveis para treinamento.

Para cada indivíduo de validação, os 1.103 indivíduos do pool completo são ordenados do maior para o menor relacionamento genômico. O rank 1 representa o vizinho mais relacionado. Depois verificamos qual é o melhor rank que permanece disponível em T25, T50, T75 e T100.

N_CENARIOS_RANK <- 30 * 276 * 4
linhas_rank <- vector(
  "list",
  N_CENARIOS_RANK
)

k <- 0L

for (rep in seq_len(30)) {
  validacao <- splits_mestre[["1"]][[rep]]$valid
  pool_index <- splits_mestre[["1"]][[rep]]$pool

  for (ind in validacao) {
    # Relacionamentos deste indivíduo da validação com o pool completo.
    g_pool <- G[ind, pool_index]

    # Ordem do mais relacionado para o menos relacionado.
    ordem_pool <- order(
      g_pool,
      decreasing = TRUE
    )

    pool_ordenado <- pool_index[ordem_pool]

    # Rank 1 = indivíduo mais relacionado do pool completo.
    rank_no_pool <- seq_along(pool_ordenado)

    nomes_rank <- setNames(
      rank_no_pool,
      pool_ordenado
    )

    for (proporcao in c(0.25, 0.50, 0.75, 1.00)) {
      treino_index <- splits_mestre[[as.character(proporcao)]][[rep]]$treino

      # Ranks dos indivíduos que pertencem ao treinamento atual.
      ranks_treino <- nomes_rank[
        as.character(treino_index)
      ]

      melhor_rank <- min(ranks_treino)

      # Indivíduo do treinamento que ocupa esse melhor rank.
      melhor_individuo <- as.integer(
        names(ranks_treino)[
          which.min(ranks_treino)
        ]
      )

      g_max_atual <- G[ind, melhor_individuo]

      melhor_individuo_pool <- pool_ordenado[1]
      g_max_pool <- G[ind, melhor_individuo_pool]

      k <- k + 1L

      linhas_rank[[k]] <- data.frame(
        repeticao = rep,
        proporcao_pool = proporcao,
        individuo_validacao = ind,
        melhor_rank_no_pool = melhor_rank,
        melhor_individuo_treino = melhor_individuo,
        melhor_individuo_pool = melhor_individuo_pool,
        g_max_atual = g_max_atual,
        g_max_pool = g_max_pool,
        perda_g_max_vs_pool = g_max_pool - g_max_atual
      )
    }
  }
}

ranking_individual <- do.call(
  rbind,
  linhas_rank
)

stopifnot(nrow(ranking_individual) == N_CENARIOS_RANK)
