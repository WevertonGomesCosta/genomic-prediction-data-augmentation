1. Objetivo

Antes de testar Data Augmentation, precisamos descobrir como o desempenho do GBLUP muda quando aumentamos a quantidade de indivíduos reais usados no treinamento.

Em cada uma das 30 repetições ajustamos o mesmo modelo usando:

  • T25 = 275 indivíduos reais;
  • T50 = 551 indivíduos reais;
  • T75 = 827 indivíduos reais;
  • T100 = 1.103 indivíduos reais.

A validação contém sempre 276 indivíduos e é exatamente a mesma para os quatro tamanhos de treinamento dentro de cada repetição.

Esta etapa responde apenas à pergunta:

Como a capacidade preditiva muda quando aumentamos o tamanho da amostra real?

A comparação formal de T25, T50 e T75 com T100 será feita somente na Etapa 5.

2. Como o GBLUP é avaliado

O ponto mais importante é que os fenótipos da validação não podem ser vistos pelo modelo.

Para isso criamos um vetor cheio de NA e copiamos para ele somente os fenótipos dos indivíduos de treinamento:

# Exemplo conceitual
y_modelo <- rep(NA_real_, length(y))
y_modelo[treino] <- y[treino]

Os genótipos continuam representados em G, mas somente o conjunto de treinamento fornece fenótipos ao BGLR.

4. Código completo dos 120 ajustes

Esta seção mostra explicitamente cada etapa do ajuste: carregamento dos objetos, mascaramento dos fenótipos, especificação do GBLUP, predição e cálculo das métricas.

if (RUN_MODELS) {
  arquivos_necessarios <- c(
    "output/01_data_objects.rds",
    "output/02_splits_mestre_30rep.rds",
    "output/03_genomic_objects.rds"
  )

  if (!all(file.exists(arquivos_necessarios))) {
    stop("Execute primeiro as Etapas 1, 2 e 3.")
  }

  if (!requireNamespace("BGLR", quietly = TRUE)) {
    stop("Instale o pacote BGLR antes de reproduzir os modelos.")
  }

  obj <- readRDS("output/01_data_objects.rds")
  splits_mestre <- readRDS("output/02_splits_mestre_30rep.rds")
  gobj <- readRDS("output/03_genomic_objects.rds")

  y <- obj$y
  G <- gobj$G

  PROPORCOES <- c(0.25, 0.50, 0.75, 1.00)
  N_ITER <- 10000
  BURN_IN <- 5000

  if (file.exists(CHECKPOINT)) {
    resultados_baseline <- read.csv(CHECKPOINT)
  } else {
    resultados_baseline <- data.frame()
  }

  for (rep in seq_len(30)) {
    for (i in seq_along(PROPORCOES)) {
      proporcao <- PROPORCOES[i]

      ja_executado <-
        nrow(resultados_baseline) > 0 &&
        any(
          resultados_baseline$repeticao == rep &
          resultados_baseline$proporcao_pool == proporcao
        )

      if (ja_executado) next

      split <- splits_mestre[[as.character(proporcao)]][[rep]]
      treino <- split$treino
      validacao <- split$valid

      # Somente os indivíduos de treinamento mantêm fenótipo observado.
      y_modelo <- rep(
        NA_real_,
        length(y)
      )

      y_modelo[treino] <- y[treino]

      stopifnot(
        all(
          is.na(
            y_modelo[setdiff(seq_along(y), treino)]
          )
        )
      )

      # GBLUP no BGLR: a matriz G entra como kernel RKHS.
      ETA <- list(
        list(
          K = G,
          model = "RKHS"
        )
      )

      set.seed(
        50000 + rep * 100 + i
      )

      ajuste <- BGLR::BGLR(
        y = y_modelo,
        ETA = ETA,
        nIter = N_ITER,
        burnIn = BURN_IN,
        verbose = FALSE
      )

      predicao <- ajuste$yHat[validacao]
      observado <- y[validacao]

      correlacao <- cor(
        observado,
        predicao
      )

      RMSE <- sqrt(
        mean(
          (observado - predicao)^2
        )
      )

      MAE <- mean(
        abs(
          observado - predicao
        )
      )

      calibracao <- lm(
        observado ~ predicao
      )

      slope_calibracao <- unname(
        coef(calibracao)[2]
      )

      nova_linha <- data.frame(
        repeticao = rep,
        proporcao_pool = proporcao,
        n_treino = length(treino),
        n_validacao = length(validacao),
        correlacao = correlacao,
        RMSE = RMSE,
        MAE = MAE,
        slope_calibracao = slope_calibracao
      )

      resultados_baseline <- rbind(
        resultados_baseline,
        nova_linha
      )

      # O checkpoint permite retomar uma execução interrompida.
      write.csv(
        resultados_baseline,
        CHECKPOINT,
        row.names = FALSE
      )
    }
  }

  write.csv(
    resultados_baseline,
    FINAL_FILE,
    row.names = FALSE
  )
}

6. Curva de aprendizado

Calculamos a média das quatro métricas e o desvio-padrão da correlação para cada tamanho de treinamento.

resumo_baseline <- aggregate(
  cbind(
    correlacao,
    RMSE,
    MAE,
    slope_calibracao
  ) ~ proporcao_pool + n_treino,
  data = baseline,
  FUN = mean
)

sd_cor <- aggregate(
  correlacao ~ proporcao_pool,
  data = baseline,
  FUN = sd
)

resumo_baseline$sd_correlacao <- sd_cor$correlacao[
  match(
    resumo_baseline$proporcao_pool,
    sd_cor$proporcao_pool
  )
]
