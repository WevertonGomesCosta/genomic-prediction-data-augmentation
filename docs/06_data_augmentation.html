1. Objetivo

Na Etapa 5 vimos que T75, com 827 indivíduos reais, já é equivalente a T100. Agora testamos se Data Augmentation permite usar ainda menos indivíduos reais.

A regra é simples: começamos com uma amostra real reduzida e criamos exatamente o número de pseudoindivíduos necessário para completar o tamanho de T100.

\[ n_{syn}=1103-n_{real}. \]

Assim:

Cenário Reais Sintéticos Total
T25 + DA 275 828 1.103
T50 + DA 551 552 1.103
T75 + DA 827 276 1.103

Os 276 indivíduos de validação continuam fora do treinamento.

2. O que é Mixup?

Um pseudoindivíduo é criado combinando dois indivíduos reais distintos do conjunto de treinamento.

Para os marcadores:

\[ \widetilde{M}=\lambda M_i+(1-\lambda)M_j. \]

Para o fenótipo:

\[ \widetilde{y}=\lambda y_i+(1-\lambda)y_j. \]

O pseudoindivíduo não é um novo indivíduo biológico observado. Ele é uma interpolação matemática entre dois indivíduos reais.

Nenhum ranking fenotípico é usado para escolher os indivíduos reais ou os dois doadores. Todos são amostrados aleatoriamente dentro do conjunto real de treinamento.

3. Como escolhemos lambda?

Testamos quatro regras:

  • lambda = 0.5: todos os pseudoindivíduos são misturas 50/50;
  • lambda ~ Beta(0.1, 0.1);
  • lambda ~ Beta(0.2, 0.2);
  • lambda ~ Beta(0.4, 0.4).

Nas distribuições Beta simétricas, a média de lambda continua próxima de 0,5, mas cada pseudoindivíduo pode receber uma intensidade de mistura diferente. Valores pequenos de alpha concentram mais probabilidade perto de 0 e 1.

6. Código completo do Data Augmentation

O bloco abaixo mostra explicitamente o procedimento usado em cada um dos 360 ajustes. Não há função auxiliar escondendo a geração dos pseudoindivíduos ou a expansão da matriz genômica.

if (RUN_MODELS) {
  arquivos_necessarios <- c(
    "output/01_data_objects.rds",
    "output/02_splits_mestre_30rep.rds",
    "output/03_genomic_objects.rds"
  )

  if (!all(file.exists(arquivos_necessarios))) {
    stop("Execute primeiro as Etapas 1, 2 e 3.")
  }

  if (!requireNamespace("BGLR", quietly = TRUE)) {
    stop("Instale o pacote BGLR antes de reproduzir os modelos.")
  }

  obj <- readRDS("output/01_data_objects.rds")
  splits_mestre <- readRDS("output/02_splits_mestre_30rep.rds")
  gobj <- readRDS("output/03_genomic_objects.rds")

  y <- obj$y
  M <- obj$M
  n <- obj$n

  p <- gobj$p
  Z <- gobj$Z
  G <- gobj$G
  denominador <- gobj$denom

  N_T100 <- 1103
  PROPORCOES <- c(0.25, 0.50, 0.75)
  N_ITER <- 10000
  BURN_IN <- 5000

  if (file.exists(CHECKPOINT)) {
    resultados_da <- read.csv(CHECKPOINT)
  } else {
    resultados_da <- data.frame()
  }

  for (ci in seq_len(nrow(configs))) {
    for (proporcao in PROPORCOES) {
      for (rep in seq_len(30)) {

        # Uma chave identifica unicamente cada modelo.
        ja_executado <-
          nrow(resultados_da) > 0 &&
          any(
            resultados_da$config_mistura == configs$config_mistura[ci] &
            resultados_da$proporcao_pool == proporcao &
            resultados_da$repeticao == rep
          )

        if (ja_executado) next

        split <- splits_mestre[[as.character(proporcao)]][[rep]]
        treino <- split$treino
        validacao <- split$valid

        M_train <- M[treino, , drop = FALSE]
        y_train <- y[treino]

        n_real <- length(treino)
        n_syn <- N_T100 - n_real

        stopifnot(n_real + n_syn == N_T100)

        # ------------------------------------------------------------
        # 1. Sorteio de dois doadores reais distintos
        # ------------------------------------------------------------
        seed_da <-
          200000 +
          ci * 10000 +
          as.integer(proporcao * 100) * 100 +
          rep

        set.seed(seed_da)

        idx1 <- sample(
          seq_len(n_real),
          n_syn,
          replace = TRUE
        )

        idx2_base <- sample(
          seq_len(n_real - 1),
          n_syn,
          replace = TRUE
        )

        idx2 <- ifelse(
          idx2_base >= idx1,
          idx2_base + 1L,
          idx2_base
        )

        stopifnot(all(idx1 != idx2))

        # ------------------------------------------------------------
        # 2. Valor de lambda de cada pseudoindivíduo
        # ------------------------------------------------------------
        if (configs$tipo_mistura[ci] == "fixed") {
          lambda <- rep(
            0.5,
            n_syn
          )
        } else {
          lambda <- rbeta(
            n_syn,
            shape1 = configs$alpha[ci],
            shape2 = configs$alpha[ci]
          )
        }

        # ------------------------------------------------------------
        # 3. Genótipos e fenótipos sintéticos
        # ------------------------------------------------------------
        M_syn <-
          lambda * M_train[idx1, , drop = FALSE] +
          (1 - lambda) * M_train[idx2, , drop = FALSE]

        y_syn <-
          lambda * y_train[idx1] +
          (1 - lambda) * y_train[idx2]

        # ------------------------------------------------------------
        # 4. Expansão da matriz G
        # ------------------------------------------------------------
        n_total_matriz <- n + n_syn
        idx_syn <- (n + 1):(n + n_syn)

        G_expanded <- matrix(
          0,
          nrow = n_total_matriz,
          ncol = n_total_matriz
        )

        # O bloco dos indivíduos reais permanece exatamente igual.
        G_expanded[
          seq_len(n),
          seq_len(n)
        ] <- G

        # Os sintéticos usam as mesmas frequências alélicas e a mesma escala.
        Z_syn <- sweep(
          M_syn,
          2,
          2 * p,
          FUN = "-"
        )

        G_syn_real <- tcrossprod(
          Z_syn,
          Z
        ) / denominador

        G_syn_syn <- tcrossprod(
          Z_syn
        ) / denominador

        G_expanded[
          idx_syn,
          seq_len(n)
        ] <- G_syn_real

        G_expanded[
          seq_len(n),
          idx_syn
        ] <- t(G_syn_real)

        G_expanded[
          idx_syn,
          idx_syn
        ] <- G_syn_syn

        G_expanded[
          cbind(idx_syn, idx_syn)
        ] <-
          G_expanded[cbind(idx_syn, idx_syn)] +
          1e-6

        stopifnot(
          max(
            abs(G_expanded - t(G_expanded))
          ) < 1e-8
        )

        stopifnot(
          max(
            abs(
              G_expanded[seq_len(n), seq_len(n)] - G
            )
          ) < 1e-12
        )

        # ------------------------------------------------------------
        # 5. Fenótipos observados pelo modelo
        # ------------------------------------------------------------
        y_modelo <- rep(
          NA_real_,
          n_total_matriz
        )

        y_modelo[treino] <- y_train
        y_modelo[idx_syn] <- y_syn

        reais_fora_treino <- setdiff(
          seq_len(n),
          treino
        )

        stopifnot(all(is.na(y_modelo[validacao])))
        stopifnot(all(is.na(y_modelo[reais_fora_treino])))
        stopifnot(sum(!is.na(y_modelo)) == N_T100)

        # ------------------------------------------------------------
        # 6. GBLUP com a matriz expandida
        # ------------------------------------------------------------
        ETA <- list(
          list(
            K = G_expanded,
            model = "RKHS"
          )
        )

        seed_bglr <-
          260000 +
          ci * 10000 +
          as.integer(proporcao * 100) * 100 +
          rep

        set.seed(seed_bglr)

        ajuste <- BGLR::BGLR(
          y = y_modelo,
          ETA = ETA,
          nIter = N_ITER,
          burnIn = BURN_IN,
          verbose = FALSE
        )

        # ------------------------------------------------------------
        # 7. Predição somente nos indivíduos reais da validação
        # ------------------------------------------------------------
        predicao <- ajuste$yHat[validacao]
        observado <- y[validacao]

        correlacao <- cor(
          observado,
          predicao
        )

        RMSE <- sqrt(
          mean(
            (observado - predicao)^2
          )
        )

        MAE <- mean(
          abs(
            observado - predicao
          )
        )

        calibracao <- lm(
          observado ~ predicao
        )

        slope_calibracao <- unname(
          coef(calibracao)[2]
        )

        nova_linha <- data.frame(
          config_mistura = configs$config_mistura[ci],
          tipo_mistura = configs$tipo_mistura[ci],
          alpha = configs$alpha[ci],
          repeticao = rep,
          proporcao_pool = proporcao,
          n_real = n_real,
          n_syn = n_syn,
          n_total_treino = n_real + n_syn,
          correlacao = correlacao,
          RMSE = RMSE,
          MAE = MAE,
          slope_calibracao = slope_calibracao,
          media_lambda = mean(lambda),
          sd_lambda = sd(lambda)
        )

        resultados_da <- rbind(
          resultados_da,
          nova_linha
        )

        # Um resultado concluído é salvo imediatamente.
        write.csv(
          resultados_da,
          CHECKPOINT,
          row.names = FALSE
        )
      }
    }
  }

  # Auditoria mínima antes de declarar a execução completa.
  chaves_da <- paste(
    resultados_da$config_mistura,
    resultados_da$proporcao_pool,
    resultados_da$repeticao,
    sep = "__"
  )

  stopifnot(nrow(resultados_da) == 360)
  stopifnot(anyDuplicated(chaves_da) == 0)
  stopifnot(all(resultados_da$n_total_treino == 1103))
  stopifnot(
    !anyNA(
      resultados_da[
        ,
        c(
          "correlacao",
          "RMSE",
          "MAE",
          "slope_calibracao"
        )
      ]
    )
  )

  write.csv(
    resultados_da,
    RAW_DA_FILE,
    row.names = FALSE
  )
}

11. TOST com correção de Nadeau-Bengio

Usamos a mesma definição da Etapa 5:

\[ \delta=0.05. \]

A diferença é considerada praticamente equivalente a T100 quando o intervalo de 90% permanece completamente dentro de [-0.05, +0.05] e os dois testes unilaterais do TOST são significativos.

Como os 30 holdouts reutilizam indivíduos da mesma população, aplicamos novamente:

\[ SE_{NB}= \sqrt{ \left( \frac{1}{R}+ \frac{n_{test}}{n_{train}} \right)s_d^2 }. \]

Aqui n_train = 1103, pois todos os cenários de DA possuem 1.103 observações de treinamento ao somar indivíduos reais e pseudoindivíduos.

DELTA <- 0.05
ALPHA <- 0.05
R <- 30
N_TEST <- 276
N_TRAIN <- 1103

resultado_equivalencia_da <- data.frame()

for (config in configs$config_mistura) {
  for (proporcao in c(0.25, 0.50, 0.75)) {

    grupo <- comparacao_t100[
      comparacao_t100$config_mistura == config &
      comparacao_t100$proporcao_pool == proporcao,
    ]

    d <- grupo$delta_cor_vs_T100

    stopifnot(length(d) == 30)

    media_d <- mean(d)
    sd_d <- sd(d)

    fator_NB <-
      1 / R +
      N_TEST / N_TRAIN

    se_NB <- sqrt(
      fator_NB * sd_d^2
    )

    gl <- length(d) - 1

    p_inferior <- 1 - pt(
      (media_d + DELTA) / se_NB,
      df = gl
    )

    p_superior <- pt(
      (media_d - DELTA) / se_NB,
      df = gl
    )

    t_critico <- qt(
      1 - ALPHA,
      df = gl
    )

    ic90_inf <- media_d - t_critico * se_NB
    ic90_sup <- media_d + t_critico * se_NB

    p_TOST <- max(
      p_inferior,
      p_superior
    )

    equivalente <-
      p_inferior < ALPHA &&
      p_superior < ALPHA &&
      ic90_inf > -DELTA &&
      ic90_sup < DELTA

    resultado_equivalencia_da <- rbind(
      resultado_equivalencia_da,
      data.frame(
        Configuracao = config,
        Proporcao = proporcao,
        N_real = unique(grupo$n_real),
        N_sintetico = unique(grupo$n_syn),
        Delta_cor_media = media_d,
        SE_NB = se_NB,
        IC90_inf = ic90_inf,
        IC90_sup = ic90_sup,
        p_TOST = p_TOST,
        Equivalente = equivalente
      )
    )
  }
}

12. Conclusão final

Sem DA, o menor conjunto equivalente a T100 foi:

\[ 827\text{ indivíduos reais}. \]

Com as quatro regras de Mixup avaliadas, T25 e T50 continuaram fora da região de equivalência. Portanto:

\[ \boxed{\text{redução adicional atribuível ao DA}=0}. \]

Sob este desenho, completar amostras aleatórias reduzidas até T100 com pseudoindivíduos Mixup não substituiu a informação dos indivíduos reais retirados. Esse resultado não significa que todo método de Data Augmentation seja ineficaz; ele descreve especificamente as estratégias de Mixup avaliadas neste tutorial.
